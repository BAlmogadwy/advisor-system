الدراســة الجامعيــة تختلــف عــن الدراســة الثانويــة فالمرحلــة الجامعيــة
تعتبر الطالب مسؤولا  مسؤولية كاملة عن حياتـه الدراسـية الج امعيـة
وتنطلــق مــن كــون الطالــب أصــبح راشــدا  . ويســتطيع أن يعتمــد علــى
نفســـه اعتمـــادا كـــاملا  فـــي تســـيير جميـــع شـــؤون حياتـــه . وعلـــى هـــذا
الأســاس فــإن لكــل طالــب مســارا  واضــحا  لحياتــه الجامعيــة تــوزع مــــن
خـلالـــه مــواده الدراسية مع مراعاة معدلاته خلال كل فصـل والمعـدل
التراكمي ،وتتميز الدراسة الجامعية بتحديد مسـتوى الطالـب بنـاء  علـى
طريقـــة جمـــع معـــدل الدراســـة خـــلال ســـنوات الجامعـــة جميعهـــا ولا
يقتصر الأمر على السنة الأخيرة كما هو الحال في المرحلة الثانويـة ،لـذا
فـــإن تقـــديرات الطالـــب خـــلال كـــل فصـــل دراســـي تصـــنع لـــه معدلـــه
التراكمي خلال جميع السـنوات وعلـى أساسـها يوضـع التقـدير النهـائي
للطالب .
لهــذا فإنــه مــن الضــروري جــدا  أن تتعــرف عزيــزي الطالــب علــى أنظمــة
الدراسة الجامعية وتستوعب أهم اللوائح والأنظمة التـي تـنظم أدائـك
الأكاديمي خلال هذه المرحلة الهامة.
يقسم العام الدراسي إلى فصلين دراسيين لا تقل مدة الفصـل عـن (15 )
أســـــبوعا  ولا يــــــدخل ضـــــمنها فتــــــرات التســـــجيل والحــــــذف والإضــــــافة
والاختبـــارات النهائيـــة وفصـــل صـــيفي اختيـــاري لا تقـــل مدتـــه عـــن ثمانيـــة
أسابيع .
❑الحد الأدنى من العبء الدراسي للفصل الرئيس في نظام الفصـلين
الدراســـيين هـــو (12 )وحـــدات دراســـية والحـــد الأعلـــى هـــو(19 )وحـــدة
دراسية .
❑الحد الأدنى مـن العـبء الدراسـي للسـنة الدراسـية فـي نظـام السـنة
الدراسية الكاملة هو (24 )وحدة دراسية والحد الأعلى هـو(39 )وحـدة
دراسية .
❑الحــد الأعلــى مــن العــبء الدراســي للفصــل الصــيفي هــو(9 )وحــدات
دراسية .
❑إذا لـــم ينجـــز الطالـــب متطلبـــات اي مقـــرر دراســـي فإنـــه يســـمح لـــه
باستكمال متطلبات المقرر بناء على توصية أستاذ المـادة، ويحصـل
علــى تقــدير غيــر مكتمــل (IC )ولا يحســب ضــمن المعــدل الفصــلي أو
التراكمي وعلى الطالب إنهاء متطلباتها في الفصل التالي وإلا تحول
إلـــى راســـب فـــي المـــادة ، ويحــــق للطالـــــب المتوقـــــع تخرجـــــه تقـــــديم
طلـــــب تســـــجيل مقـــــررات بمـــــا لا يتجـــاوز (16 )ســـاعة فـــي الفصـــــل
الدراســي ، وتنظر عمادة القبول والتسجيل في تلك الطلبات بشرط
حصول الطالب على موافقة الكلية التي يدرس فيها.
