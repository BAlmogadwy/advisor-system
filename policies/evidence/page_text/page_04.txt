عزيزي/تي الطالب/ـة
هـذادليـلـكبـيـنيـديـكفـاجـعـلـهرفـيـقـكعلـىمـدىسـنواتدراسـتـكفـيالجـامـعـة،حـتـىتـسـيربعوناللهإلـىالـنـجـاحفـي
مـســيرتـكالـدراسـيـة،ولـتـعـلـمأنالجامعـةبـكـلكـوادرهـاالبـشـريـةوجـدتلتسهيلمهمتـك،ولتأخذبـيـدكإلـىأنتمـضـيحـيـاتـك
الجامعيـةبـكـليـسـروسـهوله.
لذاحـرصـتعـمـادةالـقـبـولوالـتـسـجـيـلأنتقـدمهـذاالـدلـيـللـيـكـونلـكعـونـا  بعـداللهتعالى،ووسيلةاتصالبينـكوبـيـن
كـلـيـتـك،ولـيـضـعبـيـنأيديـكمـعـظـممـاتحـتـاجمـنمـعـرفـةحـولنـظـامالـتـعـلـيـموالـدراسـةفـيالـكـلـيـاتبالجامعـة.
حيثيـحـتـويالـدلـيـلعـلـىكـافـةالمـعـلـومـاتوالـقـنـواتالمسـاعـدةالمتعلقةبالحياةالجامعيـةالـتـيتـسـاهمفـيدعـمـك
ومـسـاندتك،وحيـثأنالإرشـادالأكاديمـييـثـبـتدائمـا  أنلهدوركبيـرفـينجاحالعمليـةالتعليمية،وفـرنـالـكبالإضافـةإلـىآلـيـة
الإرشادالأكاديمي،شـرحمـفـصـلعـنالآلـيـاتواللوائحالـدراسـيـةوالأنظمـةالإلـكـتـرونـيـةالمتاحـةلـككـطـالـبأوطالبـةإلـى
جـانـبآلـيـةالاسـتـفـادةمـنمـرشـدكالأكـاديمـي.
ويضـمالـدلـيـلجانبا  واسـعا  حـولالإرشـادالأكاديمـيحـيـثيتضمـنتـعـريـفالمصطلحاتالاساسيةبالجامعـةومدلولاتهـا،
مـفـهـومالإرشـادالأكـاديمـي،أهـدافـه،أهميتـه،محـــاوره،دورالطالب/ـةودورالمرشدالأكاديمـيفيالجامعة.
